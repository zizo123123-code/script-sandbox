```python
#!/usr/bin/env python3
"""
verify.py — أداة الفحص المستقلة للـ Manager (زيزو)
الغرض: التحقق من أي مهمة بأدلة قابلة لإعادة التنفيذ المباشر.
"""
import subprocess, sys, os

TIMEOUT_SEC = 30
MAX_STDOUT = 2000

def run(cmd, timeout=TIMEOUT_SEC):
    try:
        return subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=timeout)
    except subprocess.TimeoutExpired:
        print(f"❌ خطأ: انتهت المهلة الزمنية ({timeout}s)")
        return None

def check_commit_pushed():
    """التحقق من أن الكوميت الحالي مرفوع فعلياً على السيرفر (origin/main)"""
    r = run("git fetch origin -q && git rev-parse HEAD && git rev-parse origin/main")
    if not r or r.returncode != 0:
        return False, "تعذر قراءة الـ Refs"
    lines = r.stdout.strip().split("\n")
    if len(lines) >= 2 and lines[0] == lines[1]:
        return True, f"✅ مرفوع ومتطابق: {lines[0][:8]}"
    return False, "❌ الكوميت محلي فقط وغير مرفوع على GitHub!"

def main():
    print("=" * 60)
    print("🔍 فحص المهمة بواسطة الـ Manager (زيزو) — V0.1")
    print("=" * 60)

    # 1. آخر Commit
    r = run("git log -1 --format='%h | %an | %s'")
    print(f"📌 الكوميت : {r.stdout.strip() if r else 'فشل'}")

    # 2. فحص الرفع على GitHub
    pushed, msg = check_commit_pushed()
    print(f"📤 حالة الرفع: {msg}")

    # 3. الـ Diff
    r_diff = run("git show --stat HEAD")
    if r_diff:
        print(f"📊 التغييرات:\n{r_diff.stdout}")

    # 4. إعادة تشغيل الاختبار بنفسي (Reproducible Re-run)
    target = sys.argv[1] if len(sys.argv) > 1 else ""
    if target:
        print(f"🧪 جاري تشغيل الاختبار المستقل على: {target}")
        res = run(f"python {target}")
        if res:
            verdict = "VERIFIED ✅" if (res.returncode == 0 and pushed) else "UNVERIFIED ❌"
            print(f"🏁 القرار النهائي: Exit Code = {res.returncode} ➜ {verdict}")
            print(f"📄 مخرجات الاختبار:\n{res.stdout[-MAX_STDOUT:]}")
        else:
            print("🏁 القرار النهائي: UNVERIFIED ❌ (انتهت المهلة)")

if __name__ == "__main__":
    main()
```

# 🥊 المناظرة المعمارية الكبرى الكاملة: Antigravity vs DeepSeek (Vibe Coding Consensus)

> **سياق الجلسة:** مناظرة وحوار تقني معمق متعدد الجولات (Multi-Round Technical Duel) بين مهندس المعماريات ومحرر الـ IDE **[Antigravity]** وأيجنت الساندبوكس السحابي **[DeepSeek V4 Flash]**، بطلب وإشراف مباشر من **[زيزو]**، بهدف الوصول إلى **دستور تنفيذي نهائي 100% خالي من أي ثغرات أو تناقضات** لمنظومة الـ Multi-Agent System (V0.1).

## 📜 1. نص الرأي والمقترح المعماري الأساسي (مدخلات زيزو والفيب كودج بالكامل)

```text
بص يا زيزو، أنا فتحت الملف نفسه، وفاهم دلوقتي أنت واقف فين.
ورأيي بصراحة: متبدأش تبرمج النسخة الجديدة دلوقتي.
الفكرة عندك قوية، لكن محتاجة تتحول من "Agent بيعدل كود" إلى نظام إدارة جلسة تطوير موثّقة بين Anti-Gravity وDeepSeek.
وأنا شايف الشكل الصح كده:
1. GitHub يبقى المصدر الوحيد للحقيقة
   كل مشروع له Repo، والـAgents يتعاملوا مع الـRepo والـCommits بدل ما كل مرة يبدأوا من الصفر.
   GitHub → Snapshot الحالي → التغييرات الجديدة → Agent يفحص → اقتراح → تنفيذ → Commit → توثيق
2. الذاكرة تكون منفصلة عن الكود
   كل مشروع يبقى عنده:
   PROJECT/ ├── code/ ├── .connect/ │ ├── antigravity.md │ ├── deepseek.md │ ├── progress-antigravity.md │ └── progress-deepseek.md └── history/
   ملفات الذاكرة لا تكون بديلًا عن Git، هي فقط "ذاكرة تشغيلية" تساعد الـAgent يعرف ماذا حدث ولماذا.
3. أهم إضافة عندي: لا تعتمد على الذاكرة وحدها (Evidence-Based Verification)
   لو Agent كتب "أنا فحصت وصلحت"، ده مش دليل.
   لازم الـManager يربط كل ادعاء بحاجة قابلة للتحقق:
   (Diff + Test Code + Test Result + Commit Hash) -> لو مفيش = UNVERIFIED.
   وده متوافق جدًا مع مشكلة الـAI agents عمومًا: حتى أنظمة الـsandbox الحديثة بتفصل بين التنفيذ، السياسات، والسجل/السياق.
4. اعمل Agent Manager بينهم:
   YOU → AGENT MANAGER → (Anti-Gravity / DeepSeek) → GitHub
   الـManager يمسك Session ID, Project ID, Current Commit, Last verified state, مين فحص، مين اقترح، مين عدل، مين اختبر.
5. موضوع النسخ:
   بدل أسماء عشوائية، خليها مبنية على العملية: feature-001/ fix-002/ review-003/ مع Commit واضح.
6. البدء بـ Prototype صغير جداً V0.1 فيه 5 حاجات فقط:
   (Repo -> اختيار الأيجنت -> Snapshot للـ Commit -> تسجيل الفحص في ملفه -> التعديل ينتج Diff + Test + Commit).
```

### 🌟 المبادئ الهندسية المستخلصة والمدمجة في المنظومة:
1. **فصل المسؤوليات (Separation of Concerns):**
   - فصل (التنفيذ Execution / السياسات Policy / السياق Context) من نفس الحالة المسجلة.
2. **مبدأ الإغلاق عند الشك (Fail-Closed Execution):**
   - أي ادعاء يفتقر للدليل أو لم يتم رفعه على Git يتحول فوراً لحالة `UNVERIFIED` بدلاً من التنفيذ الأعمى.

   - أداة `verify.py` تطبع سبب عدم التحقق بدقة (مثلاً: `[verify: commit not on origin]` أو `[verify: test exit code 1]`).
4. **التصعيد بموافقة المستخدم (One-time User Escalation):**
   - إذا احتاج أيجنت تعديل ملف يخص محرر آخر، يقف فوراً ويطلب موافقة صريحة لمرة واحدة من زيزو.
5. **إعادة المحاولة كعملية مسجلة جديدة (Logged Retries):**
   - أي محاولة رفع جديدة بـ `git pull --rebase` تُسجل كعملية مستقلة لمنع الحلقات اللانهائية.

### 🌟 النقاط التي تميز هذا المخطط عن غيره:
1. **وجود "السكربت الجونيور" كقائد (Master Orchestrator):**
   - هذا هو جوهر الـ Multi-Agent Systems الحديثة؛ أنت لا تحتاج للذهاب لكل شات وتكرار كتابة التعليمات، بل سكربت جونيور يتولى صياغة وتوزيع الأوامر.
2. **استخدام Git Diff لتجنب القراءة الصفرية:**
   - بدلاً من أن يقرأ الموديل 5000 سطر كل مرة، يقرأ فقط ما بين `Commit A` و `Commit B`، مما يقضي تماماً على بطء الأداء والنسيان وسرعة انتهاء سياق المحادثة (Context Window).
3. **الفصل بين الذاكرة التشغيلية (`Progress`) والذاكرة المعرفية (`Connect`):**
   - هذا التمييز يمنع "الكروتة" لأن الموديل يعرف أين توقف في الملف الأول، وما هي البيانات الثابتة (Tokens, APIs, Endpoints) في الملف الثاني.

def load_registry():
